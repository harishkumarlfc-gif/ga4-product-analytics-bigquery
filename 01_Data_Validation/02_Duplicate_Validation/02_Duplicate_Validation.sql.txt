-- ==========================================
-- DUPLICATE VALIDATION
-- Check for duplicate Transaction-Product pairs
-- ==========================================

WITH raw_data AS (

SELECT

e.ecommerce.transaction_id,

i.item_id,

COUNT(*) AS duplicate_count

FROM
`bigquery-public-data.ga4_obfuscated_sample_ecommerce.events_*` e,
UNNEST(e.items) i

WHERE

e.event_name='purchase'

AND e.ecommerce.transaction_id IS NOT NULL

AND e.ecommerce.transaction_id!='(not set)'

GROUP BY

e.ecommerce.transaction_id,
i.item_id

)

SELECT

COUNT(*) AS unique_transaction_products,

SUM(CASE WHEN duplicate_count>1 THEN 1 ELSE 0 END) AS duplicate_transaction_products,

MAX(duplicate_count) AS max_duplicate_count

FROM raw_data;
