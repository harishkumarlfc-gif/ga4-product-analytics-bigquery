-- ==========================================
-- PRODUCT MASTER
-- Grain: One Row = One Product in One Transaction
-- ==========================================

WITH product_master AS (

SELECT

-- Transaction Details
e.ecommerce.transaction_id,

PARSE_DATE('%Y%m%d', e.event_date) AS purchase_date,

e.user_pseudo_id,

-- Product Details
i.item_id,
i.item_name,
i.item_category,
i.item_brand,

-- Sales Metrics
i.quantity,
i.price,
i.item_revenue,

-- Customer Dimensions
e.device.category AS device,
e.geo.country,
e.traffic_source.source,
e.traffic_source.medium,
e.traffic_source.name AS campaign,

-- Remove Duplicate Products Within Same Transaction
ROW_NUMBER() OVER (

PARTITION BY
e.ecommerce.transaction_id,
i.item_id

ORDER BY e.event_timestamp

) AS rn

FROM
`bigquery-public-data.ga4_obfuscated_sample_ecommerce.events_*` e,
UNNEST(e.items) i

WHERE

e.event_name = 'purchase'

AND e.ecommerce.transaction_id IS NOT NULL

AND e.ecommerce.transaction_id != '(not set)'

)

SELECT

transaction_id,
purchase_date,
user_pseudo_id,

item_id,
item_name,
item_category,
item_brand,

quantity,
price,
item_revenue,

device,
country,
source,
medium,
campaign

FROM product_master

WHERE rn = 1;
