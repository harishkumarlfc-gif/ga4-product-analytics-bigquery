WITH product_master AS (

SELECT

e.ecommerce.transaction_id,

i.item_id,
i.price,
i.quantity,
i.item_revenue,

ROW_NUMBER() OVER(

PARTITION BY
e.ecommerce.transaction_id,
i.item_id

ORDER BY e.event_timestamp

) AS rn

FROM
`bigquery-public-data.ga4_obfuscated_sample_ecommerce.events_*` e,
UNNEST(e.items) i

WHERE

e.event_name='purchase'

AND e.ecommerce.transaction_id IS NOT NULL

AND e.ecommerce.transaction_id!='(not set)'

)

SELECT

ROUND(SUM(item_revenue),2) AS total_item_revenue,

ROUND(SUM(price*quantity),2) AS total_price_x_quantity,

ROUND(SUM(price*quantity)-SUM(item_revenue),2) AS revenue_difference

FROM product_master

WHERE rn=1;
